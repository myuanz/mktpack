cmake_minimum_required(VERSION 3.26)
project(mktpack VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(cmake/dependencies.cmake)
find_package(Threads REQUIRED)
add_library(mktpack_core STATIC cpp/core/core.cpp cpp/core/block_codec.cpp cpp/core/decimal.cpp cpp/storage/result.cpp cpp/storage/dataset.cpp cpp/storage/package.cpp cpp/storage/store.cpp cpp/storage/maintenance.cpp)
target_include_directories(mktpack_core PUBLIC include)
target_link_libraries(mktpack_core PUBLIC nlohmann_json::nlohmann_json mktpack_sqlite PRIVATE libzstd_static Threads::Threads)
option(MKTPACK_BUILD_PORT "构建下载器与CLI" ON)
if(MKTPACK_BUILD_PORT)
 include(cmake/cli_dependencies.cmake)
 add_library(mktpack_port STATIC cpp/port/binance.cpp cpp/port/http.cpp cpp/port/ctp_futures.cpp cpp/port/update.cpp)
 target_include_directories(mktpack_port PRIVATE "${httplib_SOURCE_DIR}")
 target_compile_definitions(mktpack_port PRIVATE CPPHTTPLIB_MBEDTLS_SUPPORT MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
 target_link_libraries(mktpack_port PUBLIC mktpack_core mktpack_sqlite PRIVATE miniz pugixml::pugixml mbedtls mbedx509 mbedcrypto Threads::Threads)
 if(WIN32)
  target_link_libraries(mktpack_port PRIVATE crypt32)
 endif()
 add_executable(mkt cpp/cli/main.cpp)
 file(READ "${CMAKE_CURRENT_SOURCE_DIR}/docs/SKILL.md" MKTPACK_SKILL_CONTENT)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/docs/SKILL.md")
 configure_file(cpp/cli/skill.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/skill.hpp" @ONLY)
 target_include_directories(mkt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
 target_link_libraries(mkt PRIVATE mktpack_port CLI11::CLI11)
 if(MINGW)
  enable_language(RC)
  target_sources(mkt PRIVATE cpp/cli/mkt.rc)
  target_link_options(mkt PRIVATE -static -static-libgcc -static-libstdc++ -s)
 endif()
 if(SKBUILD)
  install(TARGETS mkt RUNTIME DESTINATION "${SKBUILD_SCRIPTS_DIR}")
 else()
  install(TARGETS mkt RUNTIME DESTINATION bin)
 endif()
endif()
option(MKTPACK_BUILD_PYTHON "构建Python接口" ON)
if(MKTPACK_BUILD_PYTHON)
 find_package(Python 3.12 COMPONENTS Interpreter Development.Module Development.SABIModule REQUIRED)
 execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_VARIABLE nanobind_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
 find_package(nanobind CONFIG REQUIRED)
 if(MINGW)
  set(MKTPACK_NANOBIND_OPTIONS NOMINSIZE)
 endif()
 nanobind_add_module(_native STABLE_ABI ${MKTPACK_NANOBIND_OPTIONS}
  cpp/bindings/module.cpp cpp/bindings/arrow_input.cpp cpp/bindings/arrow_output.cpp)
 if(MKTPACK_BUILD_PORT)
  target_link_libraries(_native PRIVATE mktpack_port)
  target_compile_definitions(_native PRIVATE MKTPACK_WITH_PORT)
 else()
  target_link_libraries(_native PRIVATE mktpack_core)
 endif()
 if(MINGW)
  target_link_options(_native PRIVATE -static -static-libgcc -static-libstdc++ -s)
 endif()
 install(TARGETS _native LIBRARY DESTINATION mktpack RUNTIME DESTINATION mktpack)
endif()
option(MKTPACK_BUILD_TESTS "构建测试" ON)
if(MKTPACK_BUILD_TESTS)
 enable_testing()
 add_executable(core_test tests/core.cpp)
 target_link_libraries(core_test PRIVATE mktpack_core)
 add_test(NAME core COMMAND core_test)
endif()
if(MKTPACK_BUILD_TESTS AND MKTPACK_BUILD_PORT)
 add_executable(download_test tests/download.cpp)
 target_compile_definitions(download_test PRIVATE MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
 target_link_libraries(download_test PRIVATE mktpack_core miniz pugixml::pugixml Threads::Threads)
 add_test(NAME download_lifecycle COMMAND download_test)
 add_executable(progress_test tests/progress.cpp)
 target_link_libraries(progress_test PRIVATE mktpack_core)
 add_test(NAME terminal_progress COMMAND progress_test)
endif()
